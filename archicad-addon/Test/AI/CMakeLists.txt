cmake_minimum_required(VERSION 3.17)
project(TapirWallPlacementTests LANGUAGES CXX)
enable_testing()
add_executable(test_wall_placement test_wall_placement.cpp)
target_compile_features(test_wall_placement PRIVATE cxx_std_14)
add_test(NAME wall_placement COMMAND test_wall_placement)

add_executable(test_transform_geometry test_transform_geometry.cpp)
target_compile_features(test_transform_geometry PRIVATE cxx_std_14)
add_test(NAME transform_geometry COMMAND test_transform_geometry)
add_executable(test_drawing_placement test_drawing_placement.cpp)
target_compile_features(test_drawing_placement PRIVATE cxx_std_14)
add_test(NAME drawing_placement COMMAND test_drawing_placement)
if(MSVC)
    target_compile_options(test_wall_placement PRIVATE /EHsc)
    target_compile_options(test_transform_geometry PRIVATE /EHsc)
    target_compile_options(test_drawing_placement PRIVATE /EHsc)
endif()
